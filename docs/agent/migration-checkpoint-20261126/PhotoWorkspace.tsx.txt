const base = process.env.NEXT_PUBLIC_BASE_PATH ?? "/drawloom";
const readFile = (file: File) =>
  new Promise<string>((resolve, reject) => {
    const r = new FileReader();
    r.onload = () => resolve(String(r.result));
    r.onerror = reject;
    r.readAsDataURL(file);
  });
type View =
  | "cutout"
  | "photo"
  | "features"
  | "planes"
  | "values"
  | "edges"
  | "text"
  | "hair"
  | "input"
  | "material"
  | "plane-study"
  | "structure";
export default function PhotoWorkspace() {
  const [project, setProject] = useState<PhotoProject | null>(null),
    [photo, setPhoto] = useState<HTMLImageElement | null>(null),
    [view, setView] = useState<View>("cutout"),
    [busy, setBusy] = useState(false),
    [status, setStatus] = useState(
      "Upload a photo. Crop each person before tracing.",
    ),
    [error, setError] = useState("");
  const [sourceRotation, setSourceRotation] = useState(0),
    [workspaceURL, setWorkspaceURL] = useState(''),
    [lastExport, setLastExport] = useState<{url:string;name:string} | null>(null),
    [meshOpen, setMeshOpen] = useState(false);
  const [mode, setMode] = useState<"select" | "erase" | "restore" | "pin">(
      "select",
    ),
    [brush, setBrush] = useState(24),
    [rows, setRows] = useState(40),
    [edge, setEdge] = useState(28),
    [traceMode, setTraceMode] = useState<
      "planes" | "edges" | "values" | "prepared-lines"
    >("planes"),
    [vertex, setVertex] = useState(""),
    [fontReady, setFontReady] = useState(false),
    [native, setNative] = useState(false),
    [embedded, setEmbedded] = useState(false),
    [portrait, setPortrait] = useState(true),

    cropRef = useRef<HTMLCanvasElement>(null),
    textRef = useRef<HTMLCanvasElement>(null),
    cropDrag = useRef<{ x: number; y: number } | null>(null),
    maskRef = useRef<HTMLCanvasElement | null>(null),
    paint = useRef(false),
    undo = useRef<Record<string, string[]>>({}),

    current = useRef<Figure | undefined>(figure);
  current.current = figure;
  function save(blob: Blob, name: string) {
    const reader = new FileReader();
    reader.onload = () => {
      const url = String(reader.result), a = document.createElement('a');
      setLastExport({url,name}); a.href=url; a.download=name; a.click();
    };
    reader.onerror = () => setError('The export could not be prepared.');
    reader.readAsDataURL(blob);
  }
  const update = (id: string, patch: Partial<Figure>) =>
    setProject((p) =>
      p
        ? {
            ...p,
            figures: p.figures.map((f) =>
              f.id === id ? { ...f, ...patch,
                ...(Object.hasOwn(patch, 'traceRect') && !patch.traceRect ? { rowAlignment: undefined } : {}) } : f,
            ),
          }
        : p,
    );
  async function run(action: () => Promise<void>) {
    if (busy) return;
    setBusy(true);
    setError("");
    try {
      await action();
    } catch (e) {
      setError((e as Error).message);
    } finally {
      setBusy(false);
    }
  }
  useEffect(() => {
    setEmbedded(window.parent !== window);
    const font = new FontFace("Saitamaar", `url(${base}/fonts/Saitamaar.ttf)`);
    font
      .load()
      .then((f) => {
        document.fonts.add(f);
        setFontReady(true);
      })
      .catch(() => setError("Required Saitamaar font failed to load."));
  }, []);
  async function restoreProject(text: string) {
    if (text.length > 40 * 1024 * 1024) throw Error('Choose a workspace under 40 MB.');
    const saved = validatePhotoProject(JSON.parse(text)), image = await loadImage(saved.photo);
    if (image.naturalWidth * image.naturalHeight > 25000000) throw Error('Photo exceeds 25 megapixels.');
    for (const f of saved.figures) {
      const rect = boundedRect(f.rect, image.naturalWidth, image.naturalHeight);
      if ((['x','y','width','height'] as const).some(key => rect[key] !== f.rect[key]))
        throw Error('A saved crop does not fit its original image. The project was not changed.');
    }
    epoch.current++; undo.current = {};
    setPhoto(image);
    if (saved.settings) {
      setRows(saved.settings.rows); setPortrait(saved.settings.portrait);
      setEdgeStrength(saved.settings.edgeStrength); setTraceMode(saved.settings.traceMode);
    }
    setProject(saved);
    setStatus(`Reopened ${saved.figures.length} figures, masks and plane corrections.`);
  }
  async function open(
    file: File | undefined,
    kind: "photo" | "project" | "packet",
  ) {
    if (!file) return;
    await run(async () => {
      if (file.size > 40 * 1024 * 1024)
        throw Error("Choose a file under 40 MB.");
      epoch.current++;
      undo.current = {};
      if (kind === "project") {
        await restoreProject(await file.text());
      } else {
        let data: string;
        if (kind === "packet") {
          const packet = JSON.parse(await file.text());
          const layer =
            packet.schema === "drawloom.underdrawing.v1" &&
            Array.isArray(packet.layers)
              ? packet.layers.find(
                  (layer: { kind?: string; category?: string; png?: string }) =>
                    layer.kind === "photo" && layer.category === "source",
                )
              : null;
          if (
            !layer ||
            typeof layer.png !== "string" ||
            !/^data:image\/png;base64,/.test(layer.png)
          )
            throw Error(
              "Choose a Drawloom underdrawing packet containing its original Photo source layer.",
            );
          data = layer.png;
        } else {
          if (!["image/jpeg", "image/png", "image/webp"].includes(file.type))
            throw Error("Choose JPEG, PNG or WebP.");
          data = await readFile(file);
        }
        const original = await loadImage(data),
          changed =
            sourceRotation !== 0 ||
            Math.max(original.naturalWidth, original.naturalHeight) > 2400,
          working = changed ? workingPhoto(original, sourceRotation) : null,
          workingData = working
            ? working.toDataURL(
                kind === "packet" || file.type === "image/png"
                  ? "image/png"
                  : "image/jpeg",
                0.96,
              )
            : data,
          image = working ? await loadImage(workingData) : original;
        setPhoto(image);
        setStatus("Finding the foreground faces…");
        let figures: Figure[] = [];
        try {
          figures = await suggestFigures(image);
        } catch {
          figures = [
            {
              id: crypto.randomUUID(),
              name: "Figure 1",
              rect: {
                x: 0,
                y: 0,
                width: image.naturalWidth,
                height: image.naturalHeight,
              },
            },
          ];
        }
        setProject({
          schema: "drawloom.photo-workspace.v1",
          name: file.name,
          photo: workingData,
          originalPhoto: changed ? data : undefined,
          originalRotation: sourceRotation,
          figures,
          selected: figures[0].id,
          settings: {
            ...project?.settings,
            rows,
            portrait,
            edgeStrength,
            traceMode,
          },
        });
        setStatus(
          `${figures.length} figure crops proposed. Adjust each rectangle, then prepare the selected person.`,
        );
      }
    });
  }

    const c = sourceRef.current;
    if (!c || !photo || !project) return;
    c.width = photo.naturalWidth;
    c.height = photo.naturalHeight;

    ctx.drawImage(photo, 0, 0);
    for (const f of project.figures) {
      ctx.strokeStyle = f.id === project.selected ? "#ffe063" : "#a8e5ec";
      ctx.lineWidth = f.id === project.selected ? 4 : 2;
      ctx.strokeRect(f.rect.x, f.rect.y, f.rect.width, f.rect.height);
      ctx.fillStyle = ctx.strokeStyle;

      ctx.fillText(f.name, f.rect.x + 6, f.rect.y + 26);
    }
  }, [photo, project]);
  useEffect(() => {
    const c = cropRef.current;
    if (!c || !photo || !figure) return;
    const id = ++epoch.current;
    const crop = cropPhoto(photo, figure.rect);
    c.width = crop.width;
    c.height = crop.height;
    maskRef.current = null;
    (async () => {
      let shown = await cutoutPhoto(
        crop,
        view === "photo" ? undefined : figure.mask,
        true,
      );
      if (figure.mask) {
        const m = newCanvas(crop.width, crop.height);
        m.getContext("2d")!.drawImage(
          await loadImage(figure.mask),
          0,
          0,
          m.width,
          m.height,
        );
        if (id === epoch.current) maskRef.current = m;
      }
      if (["features", "planes", "values", "edges"].includes(view)) {
        const guide = faceGuides(
          shown,
          figure,
          view as "features" | "planes" | "values" | "edges",
          edge,
        );
        if (["values", "edges"].includes(view)) shown = guide;
        else shown.getContext("2d")!.drawImage(guide, 0, 0);
      }
      if (view === "plane-study" && figure.planeGuide) {
        shown = newCanvas(crop.width, crop.height);
        shown
          .getContext("2d")!
          .drawImage(
            await loadImage(figure.planeGuide),
            0,
            0,
            shown.width,
            shown.height,
          );
      }
      if (view === "structure" && figure.structureGuide) {
        shown = newCanvas(crop.width, crop.height);
        shown
          .getContext("2d")!
          .drawImage(
            await loadImage(figure.structureGuide),
            0,
            0,
            shown.width,
            shown.height,
          );
      }
      if (view === "material" && figure.appearanceMask) {
        shown = newCanvas(crop.width, crop.height);
        const ctx = shown.getContext("2d")!;
        ctx.drawImage(
          await loadImage(figure.appearanceMask),
          0,
          0,
          shown.width,
          shown.height,
        );
        ctx.globalCompositeOperation = "source-in";
        ctx.fillStyle = "black";
        ctx.fillRect(0, 0, shown.width, shown.height);
        ctx.globalCompositeOperation = "destination-over";
        ctx.fillStyle = "white";
        ctx.fillRect(0, 0, shown.width, shown.height);
      }
      if (view === "hair" && figure.hairMask) {
        const mask = await mergeMasks(crop.width, crop.height, [
          figure.hairMask,
          figure.facialHairMask,
        ]);
        shown = newCanvas(crop.width, crop.height);
        const m = shown.getContext("2d")!;
        m.drawImage(mask, 0, 0, shown.width, shown.height);
        m.globalCompositeOperation = "source-in";
        m.fillStyle = "black";
        m.fillRect(0, 0, shown.width, shown.height);
        m.globalCompositeOperation = "destination-over";
        m.fillStyle = "white";
        m.fillRect(0, 0, shown.width, shown.height);
      }
      if (view === "input" && figure.preparedInput && figure.traceRect) {
        shown = newCanvas(crop.width, crop.height);
        const m = shown.getContext("2d")!;
        m.fillStyle = "white";

        const r = figure.traceRect;
        m.drawImage(
          await loadImage(figure.preparedInput),
          r.x,
          r.y,
          r.width,
          r.height,
        );
      }
      if (id !== epoch.current) return;
      if (
        portrait &&
        figure.pose &&
        ["plane-study", "structure", "hair", "input", "material"].includes(view)
      ) {

          shown,
          figure.traceRect ?? portraitRect(figure),
        );

        c.height = framed.height;

        return;

        ctx.lineWidth = 2;
        ctx.beginPath();
        ctx.arc(figure.seed.x, figure.seed.y, 9, 0, Math.PI * 2);

      }
    })().catch((e) => setError(e.message));
  }, [photo, figure, view, mode, edge, portrait]);

  }, [figure?.text, figure?.traceLayout, fontReady]);

  }
  const at = (event: React.PointerEvent<HTMLCanvasElement>) => {
    const c = event.currentTarget,
      r = c.getBoundingClientRect();
    return {
      x: Math.max(
        0,
        Math.min(c.width, ((event.clientX - r.left) * c.width) / r.width),
      ),
      y: Math.max(
        0,
        Math.min(c.height, ((event.clientY - r.top) * c.height) / r.height),
      ),
    };
  };
  async function prepare(f: Figure) {
    if (!photo) return f;
    const crop = cropPhoto(photo, f.rect);
    setStatus(`Separating ${f.name} with the selected-object model…`);
    const seed = f.seed ?? { x: crop.width * 0.5, y: crop.height * 0.48 },
      mask = await objectMask(crop, seed);
    let pose = f.pose;
    try {
      pose = await analyzeFigure(
        await cutoutPhoto(crop, mask.toDataURL("image/png"), true),
      );
    } catch {}
    const next = {
      ...f,
      seed,
      mask: mask.toDataURL("image/png"),
      hairMask: undefined,
      pose,
      text: undefined,
      rawText: undefined,
      traceRect: undefined,
      lineGuide: undefined,
      preparedInput: undefined,
      traceModel: undefined,
      warnings: undefined,
      traceLayout: undefined,
    };
    update(f.id, next);
    setView("cutout");
    return next;
  }
  async function trace(f: Figure) {
    if (!photo) return;
    const full = await cutoutPhoto(cropPhoto(photo, f.rect), f.mask, true);
    let region = portrait
        ? portraitRect(f)
        : { x: 0, y: 0, width: full.width, height: full.height };
    if (portrait && f.mask && f.pose) {
      const mask = await loadImage(f.mask),
        mc = newCanvas(full.width, full.height),
        cx = mc.getContext("2d")!;
      cx.drawImage(mask, 0, 0, mc.width, mc.height);
      const pixels = cx.getImageData(0, 0, mc.width, mc.height).data;
      let top = region.y;
      for (let y = 0; y < region.y; y++) {
        let count = 0;
        for (
          let x = Math.max(0, Math.floor(f.pose.bbox.cx - f.pose.bbox.size));
          x < Math.min(mc.width, f.pose.bbox.cx + f.pose.bbox.size);
          x++
        )
          if (pixels[(y * mc.width + x) * 4 + 3] > 128) count++;
        if (count > 4) {
          top = Math.max(0, y - Math.ceil(f.pose.bbox.size * 0.06));
          break;
        }
      }
      region.height += region.y - top;
      region.y = top;
    }
    const traceWidth = Math.max(128, Math.min(640, Math.round((rows * 17 * region.width) / region.height)));
    let traceRows = Math.max(8, Math.min(90, Math.round(((traceWidth / region.width) * region.height) / 17)));
    const metric = document.createElement("canvas").getContext("2d")!;
    metric.font = "16px Saitamaar";
    let eyePlan: import("@/lib/photo-workspace/plane-eye-words").PlaneEyeWordPlan | undefined;
    let rowAlignment: RowAlignment | undefined;
    if (traceMode === 'planes' && (project?.settings?.preserveFeatures ?? true)) {
      const { compilePlaneEyes } = await import('@/lib/photo-workspace/plane-eye-words');
      const frame = planeFrame(f), measure = (t: string) => Math.round(metric.measureText(t).width);
      const mode = project?.settings?.rowPhase ?? 'auto';
      if (mode === 'off') eyePlan = compilePlaneEyes(full, frame, region, traceWidth, traceRows * 17, measure);
      else {
        const baseRect = { ...region }, contentRows = Math.min(89, traceRows);
        const plans = new Map<number, typeof eyePlan>();
        const candidates: RowPhaseChoice[] = [];
        const tryPhase = async (phase: number) => {
          if (plans.has(phase)) return;
          setStatus(`Aligning ${f.name}'s whole face with native rows: ${phase} px…`);
          await new Promise<void>(resolve => requestAnimationFrame(() => resolve()));
          const plan = compilePlaneEyes(full, frame, rowPhaseRect(baseRect, contentRows, phase), traceWidth, (contentRows + 1) * 17, measure);
          plans.set(phase, plan);
          const eyes = new Map(plan.words.filter(w => w.role === 'eye-form').map(w => [w.owner, w.score]));
          const brows = new Set(plan.words.filter(w => w.role === 'brow-form').map(w => w.owner)).size;
          candidates.push({ phase, eyes: eyes.size, brows, score: eyes.size ? [...eyes.values()].reduce((a,b) => a+b,0)/eyes.size : 100 });
        };
        for (const phase of mode === 'auto' ? [0,3,6,9,12,15,16] : [mode]) await tryPhase(phase);
        candidates.sort(compareRowPhases);
        if (mode === 'auto') {
          const coarse = candidates[0].phase;
          for (const phase of [coarse-1, coarse+1]) if (phase >= 0 && phase <= 16) await tryPhase(phase);
          candidates.sort(compareRowPhases);
        }
        const chosen = candidates[0];
        rowAlignment = { phase: chosen.phase, baseRect, contentRows, candidates };
        region = rowPhaseRect(baseRect, contentRows, chosen.phase);
        traceRows = contentRows + 1;
        eyePlan = plans.get(chosen.phase)!;
        eyePlan.notes.unshift(`Whole composition aligned by ${chosen.phase} px. One row of paper preserves the original frame and scale. ${candidates.length} geometric alternatives compared; this is not a visual quality verdict.`);
      }
    }
    const cut = cropCanvas(full, region);
    const paper = cut.getContext('2d')!;
    paper.globalCompositeOperation = 'destination-over'; paper.fillStyle = 'white';
    paper.fillRect(0, 0, cut.width, cut.height); paper.globalCompositeOperation = 'source-over';
    setStatus(`Loading learned native-text filter for ${f.name}…`);
    let hair = f.hairMask;
    if (traceMode === "planes" || (project?.settings?.hairTone ?? 1) > 0) {
      if (traceMode === "planes") planeFrame(f);
      setStatus(`Resolving foreground hair over ${f.name}'s fitted head…`);
      const { hairMask } = await import("@/lib/photo-workspace/hair-tone");
      hair ??= await hairMask(full, region);
    }
    const { tracePhoto } = await import("@/lib/photo-workspace/model-trace");
    let input = cut,
      lineGuide: string | undefined,
      planeGuide: string | undefined,
      structureGuide: string | undefined,
      structureSummary: string[] | undefined,
      structureRecords: unknown,
      planeStudy:
        import("@/lib/photo-workspace/plane-structure").PlaneStudy | undefined;
    if (traceMode === "planes") {
      if (!f.pose)
        throw Error(
          "Fit the head forms and planes before selecting portrait lines.",
        );
      setStatus(
        `Fitting ${f.name}'s head planes and selecting visible form boundaries…`,
      );
      const { buildPlaneStudy } =
        await import("@/lib/photo-workspace/plane-structure");
      const study = (planeStudy = buildPlaneStudy(full, f, region, {
        hairMask: hair ? await loadImage(hair) : undefined,
        detail: project?.settings?.structureDetail ?? 0.55,
        lightBands: project?.settings?.planeBands ?? 5,
      }));
      planeGuide = study.planesCanvas.toDataURL("image/png");
      structureGuide = study.selectedLinesCanvas.toDataURL("image/png");
      structureSummary = study.summary;
      structureRecords = {
        paths: study.paths,
        planes: study.planes,
        decisions: study.decisions,
        light: study.light,
        registration: study.registration,
        eyes: study.eyes,
        shadowRegions: study.shadowRegions,
        appearanceRegions: study.appearanceRegions,
      };
      // The fitted head owns its structure. Photo contours are only proposals for
      // the visible hair/clothing outside that ownership mask, and run afterwards.
      const { photoEdges } = await import("@/lib/photo-workspace/photo-edges");
      const outside = await photoEdges(cut, {
        maxSide: 768,
        strength: edgeStrength,
        onProgress: (message) => setStatus(message),
      });
      input = newCanvas(outside.width, outside.height);
      const c = input.getContext("2d")!;
      c.drawImage(outside, 0, 0);
      c.globalCompositeOperation = "destination-out";
      c.drawImage(
        cropCanvas(study.faceMaskCanvas, region),
        0,
        0,
        input.width,
        input.height,
      );
      c.globalCompositeOperation = "multiply";
      c.drawImage(
        cropCanvas(study.selectedLinesCanvas, region),
        0,
        0,
        input.width,
        input.height,
      );
      c.globalCompositeOperation = "source-over";
      lineGuide = input.toDataURL("image/png");
    }
    if (traceMode === "edges") {
      const { photoEdges } = await import("@/lib/photo-workspace/photo-edges");
      input = await photoEdges(cut, {
        maxSide: 768,
        strength: edgeStrength,
        onProgress: (message) => setStatus(message),
      });
      lineGuide = input.toDataURL("image/png");
    }
    if (eyePlan) {
      const c = input.getContext("2d")!;
      c.fillStyle = "white";
      for (const r of eyePlan.reservedRects)
        c.fillRect(
          (r.x / traceWidth) * input.width,
          (r.y / (traceRows * 17)) * input.height,
          (r.width / traceWidth) * input.width,
          (r.height / (traceRows * 17)) * input.height,
        );
    }
    let salient: HTMLCanvasElement | undefined;
    if (
      traceMode !== "planes" &&
      (project?.settings?.preserveFeatures ?? true) &&
      f.pose
    ) {
      const { faceInk } = await import("@/lib/photo-workspace/face-ink");
      salient = faceInk(full, f.pose, region, traceWidth, traceRows * 18);
    }
    const result = await tracePhoto(input, {
      width: traceWidth,
      rows: traceRows,
      postThinInk: salient,
      mode:
        traceMode === "edges" || traceMode === "planes"
          ? "prepared-lines"
          : traceMode,
      contrast: 1,
      onProgress: (message: string) => setStatus(message),
    });
    const normalized = normalizeNativeSpacing(result.text, (text) =>

  function startPaint(event: React.PointerEvent<HTMLCanvasElement>) {
    if (
      !figure ||
      busy ||
      ["plane-study", "structure", "hair", "input", "material"].includes(view)
    )
      return;
    const p = at(event);
    event.currentTarget.setPointerCapture(event.pointerId);
    if (mode === "select") {
      update(figure.id, {
        seed: p,
        mask: undefined,
        hairMask: undefined,
        text: undefined,
        rawText: undefined,
        traceRect: undefined,
        lineGuide: undefined,
        preparedInput: undefined,
        traceModel: undefined,
        warnings: undefined,
        traceLayout: undefined,
      });
      return;
    }
    if (mode === "pin") {
      if (!figure.pose) return;
      const frame = planeFrame(figure),
        vertices = Object.values(frame.vertices).filter(
          (v) => v.visibility === "visible",
        ),
        nearest = vertices.sort(
          (a, b) =>
            Math.hypot(a.x - p.x, a.y - p.y) - Math.hypot(b.x - p.x, b.y - p.y),
        )[0];
      if (nearest) {
        setVertex(nearest.id);
        update(figure.id, { pins: { ...figure.pins, [nearest.id]: p } });
      }
      return;
    }
    if (!maskRef.current) {
      const mask = newCanvas(figure.rect.width, figure.rect.height);
      mask.getContext("2d")!.fillStyle = "white";
      mask.getContext("2d")!.fillRect(0, 0, mask.width, mask.height);
      maskRef.current = mask;
    }
    undo.current[figure.id] = [
      ...(undo.current[figure.id] ?? []),
      figure.mask ?? "",
    ].slice(-24);
    paint.current = true;
    brushPoint(p);
  }
  function brushPoint(p: { x: number; y: number }) {
    const m = maskRef.current,
      c = cropRef.current;
    if (!m || !c || !photo || !current.current) return;
    const ctx = m.getContext("2d")!;
    ctx.globalCompositeOperation =
      mode === "erase" ? "destination-out" : "source-over";
    ctx.fillStyle = "white";
    ctx.beginPath();
    ctx.arc(p.x, p.y, brush, 0, Math.PI * 2);
    ctx.fill();
    const cc = c.getContext("2d")!;
    cc.clearRect(0, 0, c.width, c.height);
    cc.drawImage(cropPhoto(photo, current.current.rect), 0, 0);
    cc.globalCompositeOperation = "destination-in";
    cc.drawImage(m, 0, 0);
    cc.globalCompositeOperation = "destination-over";
    cc.fillStyle = "white";
    cc.fillRect(0, 0, c.width, c.height);
    cc.globalCompositeOperation = "source-over";
  }

              onPointerMove={(e) => {
                if (paint.current) brushPoint(at(e));
              }}
              onPointerUp={() => {
                if (paint.current && figure && maskRef.current) {
                  paint.current = false;
                  update(figure.id, {
                    mask: maskRef.current.toDataURL("image/png"),
                    hairMask: undefined,
                    text: undefined,
                    rawText: undefined,
                    traceRect: undefined,
                    lineGuide: undefined,
                    preparedInput: undefined,
                    traceModel: undefined,
                    warnings: undefined,
                    traceLayout: undefined,
                  });
                }
              }}
            />
          </div>
          <div className="actions">
            <button
              disabled={!figure || busy}
              onClick={() => void run(previewStructure)}
            >
              Fit planes & select structure
            </button>
            <label>
              Secondary structure
              <input
                type="range"
                aria-label="Secondary plane structure"
                min="0"
                max="1"
                step="0.05"
                value={project?.settings?.structureDetail ?? 0.55}
                disabled={!project || busy}
                onChange={(e) =>
                  setProject((p) =>
                    p
                      ? {
                          ...p,
                          settings: {
                            ...p.settings,
                            rows,
                            portrait,
                            edgeStrength,
                            traceMode,
                            structureDetail: Number(e.target.value),
                          },
                        }
                      : p,
                  )
                }
              />
            </label>
            <label>
              Light bands
              <input
                type="number"
                aria-label="Plane light bands"
                min="3"
                max="9"
                value={project?.settings?.planeBands ?? 5}
                disabled={!project || busy}
                onChange={(e) =>
                  setProject((p) =>
                    p
                      ? {
                          ...p,
                          settings: {
                            ...p.settings,
                            rows,
                            portrait,
                            edgeStrength,
                            traceMode,
                            planeBands: Math.max(
                              3,
                              Math.min(9, Number(e.target.value)),
                            ),
                          },
                        }
                      : p,
                  )
                }
              />
            </label>
          </div>
          {figure?.pose && (
            <details
              open={meshOpen}
              onToggle={(e) => setMeshOpen(e.currentTarget.open)}
            >
              <summary>Fit and inspect 3D head planes</summary>
              <p className="muted">
                Orbit the fitted form. Drag a vertex on the surface to change
                the geometry used by the plane-first filter. Single-view depth
                is an estimate.
              </p>
              {meshOpen && (
                <PlanesMeshPanel
                  frame={planeFrame(figure)}
                  selected={vertex}
                  normals={false}
                  onSelect={setVertex}
                  onFace={() => {}}
                  onMove={(id, point) => {
                    update(figure.id, {
                      surfaceEdits: { ...figure.surfaceEdits, [id]: point },
                    });
                    setStatus(
                      "Head surface changed. Generate again to select lines from the revised planes.",
                    );
                  }}
                />
              )}
            </details>
          )}
          <div className="actions">
            {(
              [
                ["select", "Pick person"],
                ["erase", "Erase mask"],
                ["restore", "Restore mask"],
                ["pin", "Pin planes"],
              ] as const
            ).map(([key, label]) => (
              <button
                key={key}
                disabled={busy || !figure || (key === "pin" && !figure.pose)}
                aria-pressed={mode === key}
                onClick={() => {
                  setMode(key);
                  setView(key === "pin" ? "planes" : "cutout");
                }}
              >
                {label}
              </button>
            ))}
            <label>
              Brush
              <input
                type="range"
                aria-label="Mask brush radius"
                min="4"
                max="80"
                value={brush}
                onChange={(e) => setBrush(Number(e.target.value))}
              />
            </label>
            <button
              disabled={!figure || busy}
              onClick={() => {
                if (!figure) return;
                const old = undo.current[figure.id]?.pop();
                if (old !== undefined)
                  update(figure.id, {
                    mask: old || undefined,
                    hairMask: undefined,
                    text: undefined,
                    rawText: undefined,
                    traceRect: undefined,
                    lineGuide: undefined,
                    preparedInput: undefined,
                    traceModel: undefined,
                    warnings: undefined,
                    traceLayout: undefined,
                  });
              }}
            >
              Undo mask stroke
            </button>
          </div>
          <div className="actions">
            <button
              disabled={!figure || busy}
              onClick={() =>
                void run(async () => {
                  update(figure!.id, {
                    pose: await analyzeFigure(cropPhoto(photo!, figure!.rect)),
                  });
                  setView("planes");
                  setStatus(
                    "Single-photo estimated planes. Drag a point to correct it; depth is not validated anatomy.",
                  );
                })
              }
            >
              Analyze face & planes
            </button>
            <button
              disabled={!figure?.mask || busy}
              onClick={() => void run(() => exportImage("cutout"))}
            >
              Export transparent cutout
            </button>
          </div>
          {figure?.pose && (
            <details>
              <summary>Plane point corrections</summary>
              <select
                aria-label="Plane vertex"
                value={vertex}
                onChange={(e) => setVertex(e.target.value)}
              >
                <option value="">Choose a visible vertex</option>
                {vertices.map((v) => (
                  <option key={v.id} value={v.id}>
                    {v.id} · {v.label}
                  </option>
                ))}
              </select>
              {pin && (
                <div className="actions">
                  {(["x", "y"] as const).map((key) => (
                    <label key={key}>
                      {key}
                      <input
                        type="number"
                        aria-label={`Plane vertex ${key}`}
                        value={Math.round(pin[key])}
                        onChange={(e) =>
                          update(figure.id, {
                            pins: {
                              ...figure.pins,
                              [vertex]: {
                                ...pin,
                                [key]: Number(e.target.value),
                              },
                            },
                          })
                        }
                      />
                    </label>
                  ))}
                </div>
              )}
              <button
                onClick={() =>
                  update(figure.id, { pins: {}, surfaceEdits: {} })
                }
              >
                Reset plane corrections
              </button>
              <p className="muted">
                Single-view model estimates. Corrected points and guides persist
                in the workspace.
              </p>
            </details>
          )}
        </section>
        <section className="viewport result">
          <h2>3 · Structure → native text</h2>
          <div className="actions">
            <label>
              Rows
              <input
                type="number"
                aria-label="Native output rows"
                disabled={busy}
                min="24"
                max="90"
                value={rows}
                onChange={(e) =>
                  setRows(Math.max(24, Math.min(90, Number(e.target.value))))
                }
              />
            </label>
            <select
              aria-label="Filter source treatment"
              disabled={busy}
              value={traceMode}
              onChange={(e) => setTraceMode(e.target.value as typeof traceMode)}
            >
              <option value="planes">Planes → selected structure</option>
              <option value="edges">Photo contours only (comparison)</option>
              <option value="values">Photo values</option>
              <option value="prepared-lines">Already prepared line art</option>
            </select>
            <button
              disabled={!figure || busy || !fontReady}
              onClick={() =>
                void run(async () => {
                  const f =
                    traceMode === "prepared-lines"
                      ? figure!
                      : figure!.mask && (figure!.pose || traceMode !== "planes")
                        ? figure!
                        : await prepare(figure!);
                  await trace(f);
                })
              }
            >
              Generate AA text
            </button>
          </div>
          <div className="actions">
            <label>
              Face row alignment
              <select aria-label="Face row alignment" disabled={busy || !project || traceMode !== 'planes' || !(project.settings?.preserveFeatures ?? true)}
                value={project?.settings?.rowPhase ?? 'auto'}
                onChange={e => setProject(p => p ? { ...p, settings: { ...p.settings, rows, portrait, edgeStrength, traceMode,
                  rowPhase: e.target.value === 'auto' || e.target.value === 'off' ? e.target.value : Number(e.target.value) } } : p)}>
                <option value="auto">Fit the whole face to text rows</option>
                <option value="off">Original frame (comparison)</option>
                {Array.from({length:17}, (_,phase) => <option key={phase} value={phase}>{phase} px paper above</option>)}
              </select>
            </label>
          </div>
          {figure?.rowAlignment && <details>
            <summary>Compare face row alignments</summary>
            <p>Current: {figure.rowAlignment.phase} px. The whole portrait moves together; one extra row preserves its framing and scale. Regenerate after choosing another phase. Geometry scores help compare fits; inspect the complete art.</p>
            <div className="actions">{figure.rowAlignment.candidates.map(c => <button key={c.phase} disabled={busy}
              onClick={() => setProject(p => p ? {...p, settings:{...p.settings, rows, portrait, edgeStrength, traceMode, rowPhase:c.phase}} : p)}>
              {c.phase} px · {c.eyes} eyes · {c.brows} brows · {c.score.toFixed(2)}
            </button>)}</div>
          </details>}
          <div className="actions">
            <label>
              <input
                type="checkbox"
                aria-label="Frame face and shoulders"
                disabled={busy}
                checked={portrait}
                onChange={(e) => setPortrait(e.target.checked)}
              />
              Face & shoulders
            </label>
            <label>
              Edge strength
              <input
                type="range"
                aria-label="Learned edge strength"
                disabled={busy}
                min="0.5"
                max="2"
                step="0.1"
                value={edgeStrength}
                onChange={(e) => setEdgeStrength(Number(e.target.value))}
              />
            </label>
          </div>
          <label className="actions">
            Plane shadow tone
            <input
              type="range"
              aria-label="Plane shadow tone"
              disabled={busy || !project}
              min="0"
              max="1"
              step="0.05"
              value={project?.settings?.planeTone ?? 0.3}
              onChange={(e) =>
                setProject((p) =>
                  p
                    ? {
                        ...p,
                        settings: {
                          ...p.settings,
                          rows,
                          portrait,
                          edgeStrength,
                          traceMode,
                          planeTone: Number(e.target.value),
                        },
                      }
                    : p,
                )
              }
            />
          </label>
          <label className="actions">
            Hair shadow detail
            <input
              type="range"
              aria-label="Hair shadow detail"
              disabled={busy}
              min="0"
              max="1"
              step="0.05"
              value={project?.settings?.hairTone ?? 1}
              onChange={(e) =>
                setProject((p) =>
                  p
                    ? {
                        ...p,
                        settings: {
                          ...p.settings,
                          rows,
                          portrait,
                          edgeStrength,
                          traceMode,
                          hairTone: Number(e.target.value),
                        },
                      }
                    : p,
                )
              }
            />
          </label>
          <label className="actions">
            <input
              type="checkbox"
              aria-label={
                traceMode === "planes"
                  ? "Construct eyes from fitted forms"
                  : "Preserve eye and brow photo ink"
              }
              disabled={busy}
              checked={project?.settings?.preserveFeatures ?? true}
              onChange={(e) =>
                setProject((p) =>
                  p
                    ? {
                        ...p,
                        settings: {
                          ...p.settings,
                          rows,
                          portrait,
                          edgeStrength,
                          traceMode,
                          preserveFeatures: e.target.checked,
                        },
                      }
                    : p,
                )
              }
            />
            {traceMode === "planes"
              ? "Construct eyes from fitted forms"
              : "Preserve dark eye & brow detail"}
          </label>
          <label>
            Hair vocabulary
            <select
              aria-label="Hair tone vocabulary"
              disabled={busy}
              value={project?.settings?.hairPattern ?? "i:;"}
              onChange={(e) =>
                setProject((p) =>
                  p
                    ? {
                        ...p,
                        settings: {
                          ...p.settings,
                          rows,
                          portrait,
                          edgeStrength,
                          traceMode,
                          hairPattern: e.target.value,
                        },
                      }
                    : p,
                )
              }
            >
              <option value="flow">Follow source hair flow</option>
              <option value=":;">Sparse :; field</option>
              <option value=":">Dot columns</option>
              <option value="ﾐ">Short hatch courses</option>
              <option value="i:;">Hair stem and dot field</option>
              <option value=";:">Paired shadow field</option>
            </select>
          </label>
          <button
            disabled={busy || !project}
            onClick={() => void run(rebuildHair)}
          >
            Rebuild hair masks
          </button>
          <button
            disabled={busy || !project?.figures.some((f) => f.outlineText)}
            onClick={() => void run(refreshHair)}
          >
            Apply tone to all
          </button>
          <p className="muted">
            Fit the head forms and planes first. Visible boundaries, overlaps
            and plane values select the head structure before native text is
            articulated. Single-photo depth is estimated.
          </p>
          <div className={`paper ${native ? "native" : ""}`}>
            <canvas
              ref={textRef}
              aria-label="Learned native text preview"
              style={{ display: figure?.text ? "block" : "none" }}
            />
          </div>
          {figure?.text && (
            <>
              <div className="actions">
                <button
                  aria-pressed={native}
                  onClick={() => setNative(!native)}
                >
                  {native ? "Fit text preview" : "Native size"}
                </button>
                <button
                  disabled={busy}
                  onClick={() =>
                    save(
                      new Blob([figure.text!], {
                        type: "text/plain;charset=utf-8",
                      }),
                      `${figure.name}.txt`,
                    )
                  }
                >
                  Export native TXT
                </button>
                <button
                  disabled={busy}
                  onClick={() => void exportImage("trace")}
                >
                  Export native PNG
                </button>
                {figure.planeGuide && (
                  <>
                    <button
                      disabled={busy}
                      onClick={async () =>
                        save(
                          await (await fetch(figure.planeGuide!)).blob(),
                          `${figure.name}-plane-study.png`,
                        )
                      }
                    >
                      Export plane study
                    </button>
                    <button
                      disabled={busy}
                      onClick={() => void exportFramedStudy("planeGuide")}
                    >
                      Export framed planes
                    </button>
                  </>
                )}
                {figure.structureGuide && (
                  <>
                    <button
                      disabled={busy}
                      onClick={async () =>
                        save(
                          await (await fetch(figure.structureGuide!)).blob(),
                          `${figure.name}-plane-structure.png`,
                        )
                      }
                    >
                      Export selected structure
                    </button>
                    <button
                      disabled={busy}
                      onClick={() => void exportFramedStudy("structureGuide")}
                    >
                      Export framed structure
                    </button>
                  </>
                )}
                {figure.appearanceMask && (
                  <button
                    disabled={busy}
                    onClick={async () =>
                      save(
                        await (await fetch(figure.appearanceMask!)).blob(),
                        `${figure.name}-appearance-mask.png`,
                      )
                    }
                  >
                    Export material evidence
                  </button>
                )}
                {figure.facialHairMask && (
                  <button
                    disabled={busy}
                    onClick={async () =>
                      save(
                        await (await fetch(figure.facialHairMask!)).blob(),
                        `${figure.name}-facial-hair-mask.png`,
                      )
                    }
                  >
                    Export facial hair mask
                  </button>
                )}
                {figure.lineGuide && (
                  <button
                    disabled={busy}
                    onClick={async () =>
                      save(
                        await (await fetch(figure.lineGuide!)).blob(),
                        `${figure.name}-learned-guide.png`,
                      )
                    }
                  >
                    Export learned guide
                  </button>
                )}
                {figure.preparedInput && (
                  <button
                    disabled={busy}
                    onClick={async () =>
                      save(
                        await (await fetch(figure.preparedInput!)).blob(),
                        `${figure.name}-model-input.png`,
                      )
                    }
                  >
                    Export model input
                  </button>
                )}
              </div>
              {figure.eyePlan?.targetGuide && (
                <details>
                  <summary>Inspect eye construction and row phases</summary>
                  <p className="muted">
                    Blue lids, grey iris and black pupil show the fitted source
                    target. Green shows accepted native passages; red shows the
                    closest rejected passage. The grid is the actual native row
                    pitch.
                  </p>
                  {/* Diagnostic source and glyph targets stay read-only. */}
                  <img
                    src={figure.eyePlan.targetGuide}
                    alt="Fitted eye targets and native row phases"
                    style={{ maxWidth: "100%", background: "white" }}
                  />
                  <button
                    disabled={busy}
                    onClick={async () =>
                      save(
                        await (
                          await fetch(figure.eyePlan!.targetGuide!)
                        ).blob(),
                        `${figure.name}-eye-construction.png`,
                      )
                    }
                  >
                    Export eye construction
                  </button>
                </details>
              )}
              <details>
                <summary>Selectable native text & model</summary>
                <p>{figure.traceModel}</p>
                {figure.structureSummary?.map((text, i) => (
                  <p className="muted" key={i}>
                    {text}
                  </p>
                ))}
                <textarea
                  aria-label="Generated native text"
                  readOnly
                  value={figure.text}
                />
                {figure.warnings?.map((warning, i) => (
                  <p className="muted" key={i}>
                    {warning}
                  </p>
                ))}
              </details>
            </>
          )}
          <div className="actions">
            <button
              disabled={!figure || busy}
              onClick={() => void run(() => exportStudio())}
            >
              Export Studio project
            </button>
            {embedded && (
              <button
                disabled={!figure || busy}
                onClick={() => void run(() => exportStudio(true))}
              >
                Create project in Studio
              </button>
            )}
          </div>
          <p className="muted">
            The Studio project includes registered source, cutout, value,
            contour and plane guides, plus editable native text. Use Studio’s
            References and Tracing views for vocabulary, folds, hair and feature
            corrections.
          </p>
        </section>
      </div>
    </main>
  );
}
